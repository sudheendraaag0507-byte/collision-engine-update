#version 330 core 
layout(location = 0) in vec3 aPos;
layout(location = 1) in vec2 aLoc;
layout(location = 2) in vec3 pPos;
layout(location = 3) in float rad;
layout(location = 4) in vec3 pVel;

uniform mat4 projection ;
uniform mat4 view ;
uniform float is_box ;

out vec2 locPos;
flat out float nVel ;
flat out float box ;

void main(){
	
	mat4 mat = { vec4(2*rad , 0.0f , 0.0f , 0.0f) ,
				 vec4(0.0f , 2*rad , 0.0f , 0.0f) ,
				 vec4(0.0f , 0.0f , 2*rad , 0.0f) ,
				 vec4(pPos.x , pPos.y , pPos.z , 1.0f) 
				};

	if(is_box == 0.0f){
		gl_Position = projection * view * mat * vec4(aPos,1.0f);
	}

	else{
		gl_Position = projection * view * vec4(aPos,1.0f);
	}

	locPos = aLoc;

	box = is_box ;

	nVel = length(pVel)/1.73 ;

}
